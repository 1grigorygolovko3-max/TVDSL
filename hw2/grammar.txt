# Грамматика Funny для HW2

Ветка: LL(1), рекурсивный спуск.
Исходная грамматика приложения В приведена к виду без левой рекурсии.
Приоритет и ассоциативность зафиксированы структурой правил, а не таблицей приоритетов.

Лексер — ДКА из HW1. Пробел и комментарий `//` до конца строки пропускаются,
но сдвигают позицию. У каждого токена есть строка и столбец (с 1).

program        = { function } EOF ;
function       = IDENT "(" [ param { "," param } ] ")"
                 [ "requires" condition ]
                 "returns" param { "," param }
                 [ "ensures" condition ]
                 [ "uses" local { "," local } ]
                 statement ;
param          = IDENT ":" type ;
local          = IDENT [ ":" type ] ;
type           = "int" [ "[" "]" ] ;

statement      = assign | ifStmt | whileStmt | block | assertStmt | assumeStmt ;
block          = "{" { statement } "}" ;
ifStmt         = "if" "(" condition ")" block [ "else" block ] ;
whileStmt      = "while" "(" condition ")" [ "invariant" condition ] block ;
assertStmt     = "assert" condition ";" ;
assumeStmt     = "assume" condition ";" ;

assign         = designator "=" expr ";"
               | IDENT { "," IDENT } "=" expr ";" ;
designator     = IDENT [ "[" expr "]" ] ;
# a[i] = v  в AST сразу раскрывается в  a = set(a, i, v)

expr           = term { ("+" | "-") term } ;          # левая ассоциативность
term           = unary { ("*" | "/") unary } ;        # левая ассоциативность
unary          = "-" unary | primary ;                # правая ассоциативность
primary        = INT
               | ("length" | IDENT) "(" [ expr { "," expr } ] ")"
               | ("length" | IDENT) [ "[" expr "]" ]
               | "(" expr ")" ;

condition      = implication ;
implication    = orExpr [ "->" implication ] ;        # правая ассоциативность
orExpr         = andExpr { "or" andExpr } ;           # левая, слабее and
andExpr        = notExpr { "and" notExpr } ;          # левая, сильнее or
notExpr        = "not" notExpr | atom ;               # сильнее сравнения
atom           = "true" | "false"
               | "(" condition ")"
               | expr ("==" | "!=" | "<" | ">" | "<=" | ">=") expr ;

# Скобка в условии. Если сразу после парной ')' стоит оператор сравнения,
# скобка принадлежит выражению: (a) > 1. Иначе это скобка вокруг условия:
# (a > 1), (a > 0 and b < 2). Это единственное место, где смотрим дальше одного токена.

# Отклонения от приложения В
# - Слово function лексером выделяется, но нетерминалом грамматики не является.
#   Встретив его, парсер сообщает об ошибке и пропускает токен.
# - Формулы с `=>` и кванторы forall/exists не разбираются: в лексере HW1 нет `|`.
# - Идентификатор HW1 допускает `_`, хотя в книге шаблон [A-Za-z][A-Za-z0-9]*.
# - Пустой ввод (конец файла, пробелы или только комментарии) — ошибка.
#   Отдельные не-ASCII символы — лексические ошибки с позицией, не «пустой ввод».
# - Ветви if и тело while только блоки, как в книге, не произвольный оператор.
# - Обновление a[i] = v хранится в AST уже как присваивание вызова set.
# - Лексер не склеивает 1int в один ошибочный токен: это INT и IDENT.
# - Восстановление — режим паники: внутри оператора синхронизация по `;` или `}`,
#   внутри функции — по закрывающей `}` тела. На каждой ошибке вход сдвигается
#   хотя бы на один токен, поэтому парсер не зацикливается.
